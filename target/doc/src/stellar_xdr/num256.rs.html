25fn u256_from_pieces(hi_hi: u64, hi_lo: u64, lo_hi: u64, lo_lo: u64) -> U256 {
26    let high = (u128::from(hi_hi) << 64) | u128::from(hi_lo);
27    let low = (u128::from(lo_hi) << 64) | u128::from(lo_lo);
28    U256::from_words(high, low)
29}

32fn u256_into_pieces(u: U256) -> (u64, u64, u64, u64) {

35    let (lo_hi, lo_lo) = ((low >> 64) as u64, low as u64);
